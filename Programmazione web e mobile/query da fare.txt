{ title: { $regex: "star wars", $options: "i", $not: { $regex: "sith", $options: "i" } }, year: { $gte: 1999, $lte: 2005 } }

{
   title: {
      $regex: "star wars",
      $options: "i", 
      $not: {
         $regex: "sith",
         $options: "i"
      }
   },
   year: {
      $gte: 1999,
      $lte: 2005
   }
}
